/** Reloj controlable: now = base + tiempo real transcurrido desde setClock. TODOS los relojes del arnes (API, cliente simulado, juez, voz, L1) pasan
 *  por setClock / relojArnes, asi que el desplazamiento se aplica en un solo lugar.
 *  R11: la guarda SQL "programado_para debe ser una hora futura" usa el now() REAL de Postgres, que no se puede falsear. Para que toda hora que
 *  produce el reloj falso quede en el futuro real, el reloj se DESPLAZA un multiplo de 7 dias (mismo dia de la semana, mismas horas locales).
 *  QA_CLOCK_SHIFT_DAYS: numero = dias fijos (0 = sin desplazar, como en la R10); sin definir o "auto" = el minimo multiplo de 7 que deja el
 *  instante mas temprano del arnes (LUNES_13) al menos 1 dia despues del now() real. */
const RealDate = Date;
const DIA_MS = 86400000;
const REF_MAS_TEMPRANA = RealDate.parse("2026-10-05T19:00:00Z"); // LUNES_13: el instante mas temprano de T y de los escenarios
export function desplazamientoDias(ahora: number = RealDate.now(), env: string | undefined = process.env.QA_CLOCK_SHIFT_DAYS): number {
  if (env !== undefined && env.trim() !== "" && env.trim().toLowerCase() !== "auto") {
    const n = Number(env);
    if (!Number.isFinite(n)) throw new Error(`QA_CLOCK_SHIFT_DAYS invalido: "${env}" (numero de dias o "auto")`);
    return n;
  }
  const faltan = ahora + DIA_MS - REF_MAS_TEMPRANA;
  return faltan <= 0 ? 0 : Math.ceil(faltan / (7 * DIA_MS)) * 7;
}
export const SHIFT_DIAS = desplazamientoDias();
/** Aplica el mismo desplazamiento a un instante ISO (para el juez, para horas absolutas de escenarios). Devuelve ISO en UTC. */
export function relojArnes(iso: string): string { return new RealDate(RealDate.parse(iso) + SHIFT_DIAS * DIA_MS).toISOString(); }
let base = 0, since = 0, active = false;
export function setClock(iso: string | null) {
  if (iso === null) { active = false; return; }
  base = RealDate.parse(relojArnes(iso)); since = RealDate.now(); active = true;
}
/** R12b: copia (sin alterar la salida) de las lineas de telemetria `whatsapp_turno` que el producto escribe a stdout, para que el juez sepa que turnos fueron error_proveedor/presupuesto_agotado. */
export const telemetriaTurnos: { ts: string; resultado: string; latenciaTotalMs: number; vueltas?: number }[] = [];
{
  const w = process.stdout.write.bind(process.stdout); let resto = "";
  process.stdout.write = ((chunk: any, ...a: any[]) => {
    try {
      resto += typeof chunk === "string" ? chunk : Buffer.from(chunk).toString();
      const ls = resto.split("\n"); resto = ls.pop() ?? "";
      for (const l of ls) if (l.startsWith('{"ts"') && l.includes('"evento":"whatsapp_turno"')) { const j = JSON.parse(l); telemetriaTurnos.push({ ts: j.ts, resultado: j.resultado, latenciaTotalMs: j.latenciaTotalMs, vueltas: j.vueltas }); }
    } catch { /* best-effort */ }
    return (w as any)(chunk, ...a);
  }) as any;
}
const nowMs = () => (active ? base + (RealDate.now() - since) : RealDate.now());
// @ts-expect-error parche de Date para controlar la hora de Merida en las pruebas
globalThis.Date = class extends RealDate {
  constructor(...a: unknown[]) { if (a.length === 0) super(nowMs()); else super(...(a as [number])); }
  static now() { return nowMs(); }
};

export interface Sent { to: string; text: string | null; phoneNumberId: string; at: number }
export interface Stack {
  api: string; sent: Sent[]; pool: pg.Pool; hits: { path: string; status: number }[];
  say(from: string, body: string, opts?: { pnid?: string; id?: string }): Promise<{ status: number; ms: number; replies: string[]; raw: string; id: string; rawBody: string; sig: string }>;
  replay(d: { rawBody: string; sig: string }): Promise<number>;
  raw(payload: unknown): Promise<{ status: number; raw: string; ms: number }>;
  msgPayload(from: string, pnid: string, id: string, message: Record<string, unknown>): unknown;
  stop(): Promise<void>;
}

export const trace: { t: number; tools: { name: string; args: string }[]; text: string; lastToolResult: string | null; model: string }[] = [];
const realFetch = globalThis.fetch;
function instalarTrazaLlm() {
  globalThis.fetch = (async (input: any, init?: any) => {
    const url = typeof input === "string" ? input : input?.url ?? String(input);
    const res = await realFetch(input, init);
    if (!url.includes("openrouter.ai") || !process.env.QA_TRACE) return res;
    try {
      const reqBody = JSON.parse(typeof init?.body === "string" ? init.body : "{}");
      const msgs = reqBody.messages ?? [];
      if (process.env.QA_TRACE_FULL) (globalThis as any).__lastReq = { model: reqBody.model, messages: msgs.map((m: any) => ({ role: m.role, content: typeof m.content === 'string' ? m.content.slice(0, 80000) : m.content, tool_calls: m.tool_calls })), tools: (reqBody.tools ?? []).map((t: any) => t.function?.name) };
      const lastTool = [...msgs].reverse().find((m: any) => m.role === "tool");
      const j = await res.clone().json();
      const m = j.choices?.[0]?.message ?? {};
      trace.push({ t: Date.now(), model: j.model ?? reqBody.model, tools: (m.tool_calls ?? []).map((c: any) => ({ name: c.function?.name, args: String(c.function?.arguments ?? "").slice(0, 600) })), text: String(m.content ?? "").slice(0, 400), lastToolResult: lastTool ? String(lastTool.content).slice(0, 700) : null });
    } catch { /* traza best-effort */ }
    return res;
  }) as typeof fetch;
}
