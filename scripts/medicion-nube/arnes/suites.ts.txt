/** Medios que el arnes NO soporta (solo manda texto): se dejan como marcador y el escenario se excluye de la metrica. */
const RE_MEDIO = /\[(?:FOTO|AUDIO|STICKER|VIDEO|DOCUMENTO|LLAMADA|MENSAJE ELIMINADO|MENSAJE EDITADO|UBICACI[ÓO]N EN TIEMPO REAL)[^\]]*\]/i;
/** [PLANTILLA DE BIENVENIDA PEGADA CON DATOS LLENOS: tel, nombre, direccion, <pedido...>, pago]: el cliente pega la plantilla de la sucursal con sus datos (mismo mecanismo que G1). */
function plantillaPegada(m: string): string {
  const cuerpo = (/:\s*([^\]]*)\]?$/.exec(m)?.[1] ?? "").split(",").map((x) => x.trim()).filter(Boolean);
  const esDato = (x: string) => /^(tel[eé]fono|nombre|direcci[oó]n)$/i.test(x);
  const pago = cuerpo.find((x) => /tarjeta|efectivo/i.test(x)) ?? "efectivo";
  const pedido = cuerpo.filter((x) => !esDato(x) && x !== pago).join(", ") || "una orden de tacos de pastor";
  return `Hola, buenas noches! Para tomar su pedido:\nTeléfono: 9991234567\nNombre: Carlos Prueba\nDirección: Calle 32 #345 x 20 y 22, casa 13, Xaman Tan\nPedido: ${pedido}\nForma de pago: ${pago.replace(/^pago (con )?/i, "")}`;
}
export function limpia(t: string, zona?: { dir: string; loc: string }, proteger = false): string {
  // proteger (solo corpus-sim): los marcadores de medios y el pin ya resuelto "[Ubicación compartida por WhatsApp] lat= lng=" sobreviven al "(dato)" generico (en las suites existentes se conserva el comportamiento anterior).
  const guardados: string[] = [];
  const guardar = (x: string): string => `\u0001${guardados.push(x) - 1}\u0002`;
  if (proteger) t = t.replace(new RegExp(RE_MEDIO.source, "gi"), guardar);
  // corpus-sim (solo con `proteger`, para no cambiar pmc5/G*): marcadores nuevos ANTES de los genericos (antes cualquier [..] desconocido se volvia "(dato)")
  if (proteger) t = t.replace(/\[PLANTILLA DE BIENVENIDA PEGADA[^\]]*\]/gi, plantillaPegada)
    .replace(/\[ENLACE\]/gi, "https://ejemplo.invalid/promo").replace(/\[N[ÚU]MERO DE TARJETA\]/gi, "4111 1111 1111 1111").replace(/\[FECHA\]/gi, "12/29").replace(/\[CVV\]/gi, "123").replace(/\[N\]/g, "12");
  if (proteger && !zona) t = t.replace(/\[UBICACI[ÓO]N FUERA DE ZONA\]/gi, ZONA_AJENA["T7-049"]!.loc);
  if (zona) { const m = /lat=(-?[\d.]+) lng=(-?[\d.]+)/.exec(zona.loc); t = t.replace(/\[UBICACI[ÓO]N[^\]]*\]/gi, zona.loc).replace(/\[DIRECCI[ÓO]N[^\]]*\]/gi, zona.dir); if (m) t = t.replace(/\[LINK DE MAPS\]/gi, `https://maps.google.com/?q=${m[1]},${m[2]}`); }
  return t
    .replace(/\[UBICACI[ÓO]N[^\]]*\]/gi, LOC_T7).replace(/\[LINK DE MAPS\]/gi, "https://maps.google.com/?q=21.033777,-89.602662")
    .replace(/\[NOMBRE DEL DESARROLLO\]/gi, "Privada Los Almendros").replace(/\[NOMBRE[^\]]*\]/gi, "Carlos Prueba").replace(/\[TEL[^\]]*\]/gi, "9991234567")
    .replace(/\[DIRECCI[ÓO]N[^\]]*\]/gi, "Calle 32 #345 x 20 y 22, casa 13, Xaman Tan").replace(/\[[^\]]{1,60}\]/g, (m) => (proteger && /^\[Ubicaci[oó]n compartida/i.test(m) ? guardar(m) : RE_MEDIO.test(m) ? m : "(dato)"))
    .replace(/\u0001(\d+)\u0002/g, (_m, i) => guardados[Number(i)] ?? "");
}

/** R9: mensajes "(N min despues) ..." que llegan MUCHO despues de cerrar el pedido; el cliente simulado daba fin=true al quedar registrado y nunca los escribia (T7-006/035). */
export function partirCola(turnos: string[]): { cabeza: string[]; cola: string[] } {
  const i = turnos.findIndex((t, k) => k > 0 && /^\(\d+ min despu[eé]s\)/i.test(t));
  if (i < 0) return { cabeza: turnos, cola: [] };
  return { cabeza: turnos.slice(0, i), cola: turnos.slice(i).map((t) => t.replace(/^\(\d+ min despu[eé]s\)\s*/i, "")) };
}

export interface FiltroSim { dificultad?: number[]; categoria?: string[]; n?: number; piloto?: boolean }
/** "dificultad=3,4;categoria=pagos,pedido_kilos;n=3;piloto" -> filtro. */
export function parseFiltroSim(txt: string | undefined): FiltroSim {
  const f: FiltroSim = {};
  for (const par of (txt ?? "").split(";").map((x) => x.trim()).filter(Boolean)) {
    const [k, v = ""] = par.split("=");
    if (k === "dificultad") f.dificultad = v.split(",").map(Number).filter((x) => x >= 1 && x <= 4);
    else if (k === "categoria") f.categoria = v.split(",").map((x) => x.trim()).filter(Boolean);
    else if (k === "n") f.n = Math.max(1, Number(v) || 1);
    else if (k === "piloto") f.piloto = v === "" || v === "1" || v === "true";
  }
  return f;
}
const hashId = (id: string): number => { let h = 2166136261; for (const ch of id) { h ^= ch.charCodeAt(0); h = Math.imul(h, 16777619); } return h >>> 0; };
/** Piloto estratificado (~`objetivo` escenarios): por categoria, al menos min(3, tamano) y el resto proporcional; dentro de la categoria, alterna dificultades (de la mas dificil a la mas facil) con orden determinista por hash del id. */
export function seleccionPiloto(es: any[], objetivo = 100): any[] {
  const porCat = new Map<string, any[]>(); for (const e of es) (porCat.get(e.categoria) ?? porCat.set(e.categoria, []).get(e.categoria)!).push(e);
  const total = es.length; const out: any[] = [];
  for (const [, xs] of [...porCat].sort((a, b) => a[0].localeCompare(b[0]))) {
    const cupo = Math.min(xs.length, Math.max(3, Math.round((objetivo * xs.length) / total)));
    const porDif = new Map<number, any[]>(); for (const e of [...xs].sort((a, b) => hashId(a.id) - hashId(b.id))) (porDif.get(e.dificultad) ?? porDif.set(e.dificultad, []).get(e.dificultad)!).push(e);
    const niveles = [...porDif.keys()].sort((a, b) => b - a); const elegidos: any[] = [];
    for (let r = 0; elegidos.length < cupo && r < xs.length; r++) for (const d of niveles) { const e = porDif.get(d)![r]; if (e && elegidos.length < cupo) elegidos.push(e); }
    out.push(...elegidos);
  }
  return out;
}
export interface ResumenSim { total: number; activos: number; pendientes: number; medios: number; seleccionados: number }
/** Carga y valida la FORMA del corpus (sin LLM ni stack). Lanza si el JSON no cumple el contrato minimo. */
export function cargarCorpusSim(ruta = rutaCorpusSim()): any[] {
  const d = JSON.parse(readFileSync(ruta, "utf8")); const es: any[] = d.escenarios;
  if (!Array.isArray(es) || es.length === 0) throw new Error("corpus-sim: sin escenarios");
  const ids = new Set<string>();
  for (const e of es) {
    for (const k of ["id", "turnos_cliente", "comportamiento_esperado", "que_no_debe_hacer", "estado", "categoria", "dificultad", "resultado_esperado", "reloj"]) if (e[k] === undefined) throw new Error(`corpus-sim ${e.id ?? "?"}: falta ${k}`);
    if (ids.has(e.id)) throw new Error(`corpus-sim: id repetido ${e.id}`); ids.add(e.id);
    if (!Array.isArray(e.turnos_cliente) || e.turnos_cliente.length === 0 || e.turnos_cliente.some((t: unknown) => typeof t !== "string")) throw new Error(`corpus-sim ${e.id}: turnos_cliente invalido`);
    if (!RELOJ_SIM[e.reloj]) throw new Error(`corpus-sim ${e.id}: reloj desconocido ${e.reloj}`);
    if (!["comanda", "escalar", "sin_comanda"].includes(e.resultado_esperado?.resultado)) throw new Error(`corpus-sim ${e.id}: resultado_esperado invalido`);
    if (e.rafagas?.some((r: number[]) => r.length !== 2 || r[0]! < 0 || r[1]! >= e.turnos_cliente.length || r[0]! > r[1]!)) throw new Error(`corpus-sim ${e.id}: rafagas fuera de rango`);
  }
  return es;
}
const esMedio = (e: any): boolean => e.turnos_cliente.some((t: string) => RE_MEDIO.test(t));
export function resumenSim(es: any[], filtro: FiltroSim = {}): ResumenSim & { porCategoria: Record<string, number> } {
  const sel = filtrarSim(es, filtro); const porCategoria: Record<string, number> = {}; for (const e of sel) porCategoria[e.categoria] = (porCategoria[e.categoria] ?? 0) + 1;
  return { total: es.length, activos: es.filter((e) => e.estado !== "pendiente_decision").length, pendientes: es.filter((e) => e.estado === "pendiente_decision").length, medios: es.filter(esMedio).length, seleccionados: sel.length, porCategoria };
}
/** Aplica el filtro. Los pendientes_decision y los de medios se CONSERVAN en la lista (para listarlos); la seleccion de piloto/n solo reparte entre los medibles. */
export function filtrarSim(es: any[], f: FiltroSim): any[] {
  let xs = es.filter((e) => (!f.dificultad?.length || f.dificultad.includes(e.dificultad)) && (!f.categoria?.length || f.categoria.includes(e.categoria)));
  const medible = (e: any) => e.estado !== "pendiente_decision" && !esMedio(e);
  if (f.piloto) { const base = seleccionPiloto(xs.filter(medible)); xs = base; }
  if (f.n) { const cuenta = new Map<string, number>(); xs = [...xs].sort((a, b) => hashId(a.id) - hashId(b.id)).filter((e) => { const c = cuenta.get(e.categoria) ?? 0; if (c >= f.n!) return false; cuenta.set(e.categoria, c + 1); return true; }); }
  return xs.sort((a, b) => a.id.localeCompare(b.id));
}

function hardSim(e: any): Job["hard"] | undefined {
  const r = e.resultado_esperado ?? {}; const pasos: NonNullable<Job["hard"]>[] = [];
  if (r.pedidos === 0 || r.resultado !== "comanda") pasos.push((c) => (nuevos(c).length ? [`no debia crear pedido y creo ${nuevos(c).length} (total ${nuevos(c)[0].total})`] : []));

    pasos.push((c) => (nuevos(c).length ? [] : ["debia crear el pedido y no creo ninguno"]));

}
